async function checkStatus() { if (!currentRequestId) { return; } try { const response = await fetch( "/demo-status/" + encodeURIComponent(currentRequestId), { cache: "no-store" } ); if (!response.ok) { return; } const data = await response.json(); console.log("Demo status:", data); // ========================= // APPROVED // ========================= if (data.status === "approved") { clearInterval(pollingTimer); pollingTimer = null; document .getElementById("waitingOverlay") .classList.remove("active"); // Page 1 → Page 2 if (Number(data.page) === 1) { document .getElementById("page1") .classList.add("hidden"); document .getElementById("page2") .classList.remove("hidden"); document .getElementById("status1") .textContent = ""; document .getElementById("message2") .focus(); currentRequestId = null; currentPage = 2; return; } // Page 2 → Final approval if (Number(data.page) === 2) { currentRequestId = null; showResult( "approved", "Demo completed successfully." ); return; } } // ========================= // DECLINED // ========================= if (data.status === "declined") { clearInterval(pollingTimer); pollingTimer = null; document .getElementById("waitingOverlay") .classList.remove("active"); // Page 1 → Page 1 again if (Number(data.page) === 1) { const input = document.getElementById("message1"); input.value = ""; document .getElementById("status1") .textContent = "Please enter a new demo message."; document .getElementById("page1") .classList.remove("hidden"); input.focus(); currentRequestId = null; currentPage = 1; return; } // Page 2 → Page 2 again if (Number(data.page) === 2) { const input = document.getElementById("message2"); input.value = ""; document .getElementById("status2") .textContent = "Please enter a new demo message."; document .getElementById("page2") .classList.remove("hidden"); input.focus(); currentRequestId = null; currentPage = 2; return; } } } catch (error) { console.error( "Status check failed:", error ); } }
